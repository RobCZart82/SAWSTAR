set(IPLUG2_DIR "${PROJECT_SOURCE_DIR}/third_party/iPlug2")
if(NOT EXISTS "${IPLUG2_DIR}/iPlug2.cmake")
  message(FATAL_ERROR "Run python scripts/bootstrap-plugin.py before configuring the plugin")
endif()
if(NOT EXISTS "${IPLUG2_DIR}/Dependencies/IPlug/VST3_SDK/public.sdk/source/main/pluginfactory.cpp")
  message(FATAL_ERROR "The pinned VST3 SDK is missing. Run python scripts/bootstrap-plugin.py")
endif()
set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "Do not install plugins during build")
set(IGRAPHICS_BACKEND NANOVG)
if(APPLE)
  set(IGRAPHICS_RENDERER METAL)
else()
  set(IGRAPHICS_RENDERER GL2)
endif()
include(${IPLUG2_DIR}/iPlug2.cmake)
find_package(iPlug2 REQUIRED)
iplug_add_plugin(SAWSTAR
  FORMATS VST3
  SOURCES SAWSTAR.cpp SAWSTAR.h config.h ../gui/Controls/PageButton.h
  LINK SAWSTAR::Foundation sawstar_engine)
set_target_properties(SAWSTAR-vst3 PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/out"
  LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/out")
# Keep the SDK's Windows bundle layout (the generic properties above are macOS only).
if(WIN32)
  if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
    set(_sawstar_win_arch "arm64-win")
  else()
    set(_sawstar_win_arch "x86_64-win")
  endif()
  target_compile_definitions(SAWSTAR-vst3 PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  set_target_properties(SAWSTAR-vst3 PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/out/SAWSTAR.vst3/Contents/${_sawstar_win_arch}"
    LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/out/SAWSTAR.vst3/Contents/${_sawstar_win_arch}")
endif()

# Embed an unchanged OFL font for the About wordmark; no PNG texture is needed.
set(_font "${PROJECT_SOURCE_DIR}/assets/fonts/Orbitron-Bold.ttf")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_font}")
file(READ "${_font}" _font_hex HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _font_bytes "${_font_hex}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AboutFont.h"
  "#pragma once\nnamespace sawstar::gui { inline unsigned char AboutFont[]={${_font_bytes}}; }\n")
target_include_directories(SAWSTAR-vst3 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

if(APPLE)
  target_sources(SAWSTAR-vst3 PRIVATE ../gui/NativeImport.mm)
else()
  target_sources(SAWSTAR-vst3 PRIVATE ../gui/NativeImport.cpp)
  target_link_libraries(SAWSTAR-vst3 PRIVATE comdlg32)
endif()

# Patch a build-directory copy only; keep the pinned dependency untouched.
set(_entry_source "${IPLUG2_DIR}/IGraphics/Controls/ITextEntryControl.cpp")
file(READ "${_entry_source}" _entry)
string(FIND "${_entry}" "case kVK_RETURN: CommitEdit(); break;" _enter_pos)
if(_enter_pos EQUAL -1)
  message(FATAL_ERROR "Review the iPlug2 text-entry Enter patch after dependency changes")
endif()
string(REPLACE "case kVK_RETURN: CommitEdit(); break;" "case (kVK_RETURN | 0x8000):
    case kVK_RETURN: CommitEdit(); return true;" _entry "${_entry}")
string(REPLACE "case kVK_ESCAPE: DismissEdit(); break;" "case kVK_ESCAPE: DismissEdit(); return true;" _entry "${_entry}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SAWSTARTextEntry.cpp" "${_entry}")
get_target_property(_graphics_sources iPlug2::IGraphics INTERFACE_SOURCES)
list(FIND _graphics_sources "${_entry_source}" _entry_index)
if(_entry_index EQUAL -1)
  message(FATAL_ERROR "Cannot locate iPlug2 text-entry source")
endif()
list(REMOVE_ITEM _graphics_sources "${_entry_source}")
list(APPEND _graphics_sources "${CMAKE_CURRENT_BINARY_DIR}/SAWSTARTextEntry.cpp")
set_property(TARGET iPlug2::IGraphics PROPERTY INTERFACE_SOURCES "${_graphics_sources}")

execute_process(COMMAND git rev-parse --short=7 HEAD WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
  OUTPUT_VARIABLE SAWSTAR_BUILD_ID OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT SAWSTAR_BUILD_ID)
  set(SAWSTAR_BUILD_ID "local")
endif()
string(JSON SAWSTAR_METADATA_RELEASE_DATE ERROR_VARIABLE SAWSTAR_DATE_ERROR GET "${SAWSTAR_METADATA}" release_date)
if(SAWSTAR_DATE_ERROR)
  set(SAWSTAR_METADATA_RELEASE_DATE "")
endif()
set(SAWSTAR_RELEASE_DATE "${SAWSTAR_METADATA_RELEASE_DATE}" CACHE STRING "Public release date (YYYY.MM.DD.); empty for development builds")
if(SAWSTAR_RELEASE_DATE)
  if(NOT SAWSTAR_RELEASE_DATE MATCHES "^[0-9][0-9][0-9][0-9]\\.[0-9][0-9]\\.[0-9][0-9]\\.$")
    message(FATAL_ERROR "SAWSTAR_RELEASE_DATE must use YYYY.MM.DD.")
  endif()
  set(SAWSTAR_DISPLAY_VERSION "${PROJECT_VERSION}")
  set(SAWSTAR_DATE_LABEL "Release")
  set(SAWSTAR_BUILD_DATE "${SAWSTAR_RELEASE_DATE}")
else()
  set(SAWSTAR_DISPLAY_VERSION "${PROJECT_VERSION} Pre Release")
  set(SAWSTAR_DATE_LABEL "Build")
  string(TIMESTAMP SAWSTAR_BUILD_DATE "%Y.%m.%d." UTC)
endif()
configure_file(BuildVersion.h.in BuildVersion.h @ONLY)

  set_property(SOURCE SAWSTAR.cpp APPEND PROPERTY COMPILE_OPTIONS -Wall -Wextra -Wpedantic)
endif()
